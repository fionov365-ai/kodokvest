### recursion
def fact(n):
    if n <= 1:
        return 1
    return n * fact(n - 1)
print(fact(5))

### mixed
words = ["кот", "слон", "ёж"]
longest = ""
for w in words:
    if len(w) > len(longest):
        longest = w
print(longest)
print(", ".join(words))
counts = {}
for w in words:
    counts[len(w)] = counts.get(len(w), 0) + 1
print(counts)
